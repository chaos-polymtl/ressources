\begin{frame}{\textbf{Volume-Averaged Navier-Stokes (VANS) equations}}
	\visible<1->
	{\begin{shaded}
			The \textbf{Volume-Averaged Navier-Stokes} are used to simulate flows with spatially ($\bm{x}$) and temporally ($t$) varying void fraction ($\epsilon$)
	\end{shaded}}
	
	\begin{minipage}[t]{0.48\linewidth}
		%\visible<2->
		{
			\begin{block}{Used in the following models:}
				\begin{itemize}
					\item Two-fluid models
					\item Unresolved CFD-DEM
					\item Porous media flow
				\end{itemize}
			\end{block}
		}
	\end{minipage}%
	\hfill%
	%
	\begin{minipage}[t]{0.48\linewidth}
		%\visible<3->
		{
			\begin{block}{For many applications:}
				\begin{itemize}
					\item Fluidized and spouted-bed reactors
					\item Free surface flows in containers
					\item DNS of particle-laden flow
					\item Mixing in stirred tanks 
					\item Flow in porous media
				\end{itemize}
			\end{block}
		}
	\end{minipage}
\end{frame}


\begin{frame}{\textbf{VANS equations}}
\begin{block}{Mass conservation:}
\[
\partial_t \varepsilon + \partial_i (\varepsilon u_i) = 0
\]
\end{block}
\begin{block}{Momentum conservation:}
	\[
	\partial_t \left(\varepsilon u_i\right) + \partial_j (\varepsilon u_i u_j) = \frac{\varepsilon}{\rho} \partial_i p + \partial_j \left(\epsilon \tau_{ij}\right) + \epsilon g_i
	\]
	\end{block}
	\begin{itemize}
	\item $\varepsilon$ the void fraction
	\item $u_i$ the velocity
	\item $\rho$ the density
	\item $p$ the pressure
	\item $\tau_{ij}$ the deviatoric stress tensor
	\item $g_i$ the body acceleration per unit volume (e.g. gravity)
	\end{itemize} 


\end{frame}

	\frametitle{\textbf{A simplified representation of this problem}}
	
Without loss of generality, let's assume steady state and $\varepsilon=\varepsilon(z)$ such that:
\[
\varepsilon(z) = \begin{cases}
	1 & \text{if } z < z_0 \\
	0.5 & \text{if } z \in[z_0, z_1] \\
	1 & \text{if } z > z_1
\end{cases}
\]
And that the velocity at the inlet is uniform in the $xy$ plane and constant ($u$). Then, from mass conservation, we have:
\[
\partial_t \varepsilon + \partial_z (\varepsilon u_z) = 0
\]
Since we are in steady state, we have $\partial_t \varepsilon = 0$, and thus:
\[
\partial_z (\varepsilon u_z) = 0
\]
This means that $u_z$ is:
\[
	\varepsilon(z) = \begin{cases}
		u & \text{if } z < z_0 \\
		2u & \text{if } z \in[z_0, z_1] \\
		u & \text{if } z > z_1
	\end{cases}
\]

\end{frame}

\section{Origin of the equations}

\begin{frame}
\frametitle{\textbf{Origin of the VANS equations}}
\begin{block}{Reference}
There are many ways to derive the VANS equations, but they all rely on the same idea: \textbf{volume averaging}. Here I will follow Anderson \& Jackson's approach %\cite{anderson_fluid_1967}
. However, alternative/similar demonstration exist notably in the book by Balachandar %\cite{balachandar2024fundamentals}
 Crowe \textit{et al.} %\cite{crowe1998multiphase}
 Prosperetti and Tryggvason %\cite{prosperetti2009computational
  etc.
\end{block}

\begin{block}{Small Warning}
These kind of demonstration is something we take for granted, but it is not trivial. It requires a good understanding of the physics and mathematics behind the equations. I do not feel people have reached clear conclusion on the best way to derive these equations. I will try to give you the main ideas.
\end{block}

\end{frame}

\begin{frame}{\textbf{Volume averaging}}

	{\begin{shaded}
\textbf{Main idea:} Replace the pointwise mechanical variables (velocity, pressure, etc.) by local mean variables obtained by point variables over region which contain many particles, but are still small compared with the scale of \textit{macroscopic} variation from point to point in the system.
	\end{shaded}}

\textbf{Obviously this is very vague, let's try to be more precise.}

\end{frame}

\begin{frame}{\textbf{Volume averaging}}
Let $g(r)$ be a function defined for $r>0$ with the following properties:
\begin{itemize}
	\item $g(r)>0\; \forall\; r$
	\item $g(r)$ decreases monotonically with $r$
	\item $g(r)$ possess derivatives $g^{n}(r)$ of all orders for each value of $r$
	\item $\int_{V_\infty} g^{n}(r) \mathrm{d}V$ exists for all values of $n$. $V_\infty$ is the entire volume of the system. 
	\item \textbf{Theoretically it should have a compact support, i.e. $g(r)=0$ for $r>R$ where $R$ is a finite radius.}
\end{itemize}

From there:
\[
\int_{V_\infty} g(r) \mathrm{d}V = \int_0^{\infty} g(r) r^2 \mathrm{d}r 
\]

And we normalize $g(r)$ such that:
\[
\int_{V_\infty} g(r) \mathrm{d}V = 1
\]

\textbf{$g(r)$ is a weighting function} 

\end{frame}


\begin{frame}{\textbf{Volume averaging}}

	{\begin{shaded}
\textbf{$g(r)$ is a weighting function:} It is used to form local averages of system point variables.
			\end{shaded}}

In practice, this function will have a compat support. We will speak of the radius $r_0$ of the weighting function as:
\[
4\pi \int_0^{r_0} g(r) r^2 \mathrm{d}r = 4\pi \int_{r_0}^{\infty} g(r) r^2 \mathrm{d}r = \frac{1}{2}
\]

\textbf{Note: } This radius is not a \textit{physical} radius, but rather a measure of the size of the region over which we average. 
	
\end{frame}



\begin{frame}{\textbf{Void fraction definition}}
	
	{\begin{shaded}
	Using the weighting function $g(r)$, we can define the void fraction $\varepsilon$.
	\end{shaded}}

	\[
\varepsilon(x_i,t) = \int_{V_{f,\infty}} g(|x_i-y_i|) \mathrm{d}V_y
	\]
where $V_{f,\infty}$ indicates that the integral is taken over all points $y_i$ occupied by the fluid at time $t$. 

\end{frame}


\begin{frame}{\textbf{Particle density}}

Let $V_p$ be the volume of one particle. All particles are assumed to be identical. The local mean particle density (number of particles per unit volume) is:
\begin{align*}
V_p n_p(x_i,t) &= 1 - \varepsilon(x_i,t) \\
               &= 1 - \int_{V_{f,\infty}} g(|x_i-y_i|) \mathrm{d}V_y \\
							 &= \int_{V_{\infty}} g(|x_i-y_i|) \mathrm{d}V_y - \int_{V_{f,\infty}} g(|x_i-y_i|) \mathrm{d}V_y 
\end{align*}

If we denote $V_{s,\infty}$ as the set of all points occupied by the solid phase (points inside the particles), we can write:
\begin{align*}
V_{s,\infty} = V_\infty  - V_{f,\infty}
\end{align*}
\end{frame}

\section{A straightforward implementation}

\begin{frame}{\textbf{Preamble: Problem definition for Navier-Stokes}}
	Consider a $d$-dimensional domain $\Omega$ and the following problem subjected to homogeneous Dirichlet boundary conditions: 
	%\vspace{0.1cm}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			% \vspace*{-\baselineskip}
			\begin{framed}
				Find a vector ${u}_i$ and a scalar $p$ in $\Omega \rightarrow \mathbb{R}$ satisfying:
				\begin{align*}
					\partial_i u_i &= 0 \\
					\partial_{t} {u}_i + ({u}_j \partial_j ) u_i + \partial_i p - \nu \partial_j \partial_j {u}_i &= 0
					% \bm{u} &= 0 \ \textrm{ in } \partial\Omega
					% \textrm{ in } \Omega
				\end{align*}
			\end{framed}
		\end{column}
		\begin{column}{0.4\textwidth}
			\vspace*{-\baselineskip}
			\begin{center}
				\includegraphics[scale = 0.13]{images/domain_3d.png}
			\end{center}
		\end{column}
	\end{columns}
	
	\vspace{0.4cm}
	\begin{itemize}
		\item This is a coupled problem for velocity (${u}_i$) and pressure ($p$).
		\item This is a non-linear problem due to the second term in the momentum equation.
	\end{itemize}    
\end{frame}


\begin{frame}
	\frametitle{\textbf{Premable: Variational formulation}}
	%\fontsize{10.5}{11}\selectfont
	We multiply by the test functions $v_i$ and $q$, and integrate over the domain using the notation $(a, b)_{\Omega} = \int a b \ \text{d}\Omega$:
	\begin{align*}
		\left( q, \partial_i \cdot u_i \right)_{\Omega} + (v_i, \partial_t u_i + (u_j \partial_j) \bm{u} + \underbrace{\partial_i p}_{1} - \underbrace{\nu \partial_j \partial_j u_i}_{2})_{\Omega} = 0
	\end{align*}
	We integrate by parts terms $1$ and $2$:
	\begin{align*}
		({v}_i, \partial_i p)_{\Omega} &= - (\partial_i  {v}_i, p)_{\Omega} + ( n_i  {v}_i, p)_{\partial\Omega} \\
		({v}_i, -\nu \partial_j \partial_j {u}_i)_{\Omega} &= \nu (\partial_i {v}_j,  \partial_i {u}_j)_{\Omega} - (n_i {v}_j, \nu \partial_i {u}_j)_{\partial\Omega} 
	\end{align*}
	If we assume Dirichlet B.Cs or zero stress conditions on $\partial\Omega$:
	\begin{align*}
		( {n}_i {v}_i, p)_{\partial\Omega} - ({v}_i {n}_j, \nu \partial_i {u}_j)_{\partial\Omega} = 0
	\end{align*}
	%which is an outlet boundary condition where the normal stress is zero. Approximately empose an outlet boundary condition with a zero average pressure.
	Writing everything together we get the final weak form:
	\begin{align*}
		F({u}_i, p) := \left( q, \partial {u}_i \right)_{\Omega} + (v_i, \partial_t {u}_i)  + {({v}_i, ({u}_j \partial_j) {u}_i)_{\Omega}} 
		- (\partial_i  {v}_i, p)_{\Omega} + \nu (\partial_i {v}_j, \partial_i {u}_j)_{\Omega}  = 0
	\end{align*}
	%\visible<2->{
	%	\textcolor{maincolor}{$\longrightarrow$ due to the non linearity we use \textbf{Newton's method} to solve this problem.}}
\end{frame}



\begin{frame}{\textbf{VANS: Mass conservation}}
For mass conservation, we have:
\begin{align*}
\partial_t \epsilon + \partial_i (\epsilon u_i) = 0
\end{align*}
We multiply by the test function $q$ and integrate over the domain:
\begin{align*}
	\pp{q,\partial_t \varepsilon}_\Omega + \pp{q, \varepsilon \partial_i u_i}_\Omega + \pp{q, u_i \partial_i \varepsilon}_\Omega + \underbrace{\pp{\tau_{\vm{u}} \partial_i q, \mathcal{R}_{i}}_{\Omega_e}}_{\mathrm{PSPG}} = 0
\end{align*}

The last term is our traditional PSPG term to enable the use of equal-order interpolation. The expression of the strong residual is:
\begin{align*}
	\mathcal{R}_{i} &= \varepsilon \partial_t u_i + \varepsilon {u_j} \partial_j \cdot {u_i} 
	+ \frac{1}{\rho} \bra{\varepsilon \partial_i p - \nu \varepsilon \partial_j \partial_j {u_i} + \partial_i {u_j} \partial_j \varepsilon}
\end{align*}
\end{frame}

\section{Conclusion}

\begin{frame}{\textbf{Conclusion}}
\begin{block}{Summary}
	\begin{itemize}
		\item The VANS equations are used to model flows with spatially and temporally varying void fraction.
		\item They are derived from the Navier-Stokes equations using volume averaging.
		\item By themselves, they are challenging to solve using the finite element method.
	\end{itemize}
\end{block}
\begin{block}{Topics I did not discuss}
	\begin{itemize}
		\item Defining the void fraction from Lagrangian particles.
		\item Force term that arise due to the interaction of the fluid with the particles.
	\end{itemize}
\end{block}
\textbf{Remember that the VANS equations are physical equations and, consequently, should not be correlated with the numerical discretization.}
\end{frame}
